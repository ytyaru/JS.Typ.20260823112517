```javascript
const decimal = fig => {
    if (15 < fig) {throw new RangeError(`figは15以下であるべきです。`)}
    const scale = 10 ** fig;
    const min = Number.MIN_SAFE_INTEGER / scale;
    const max = Number.MAX_SAFE_INTEGER / scale;
    return {min, max, valid(v){return this.min<=v && v<=this.max}};
};
const d = decimal(2);
console.assert(true===d.valid(99999.99)); // 少数2桁を維持できる
console.assert(false===d.valid(99999999999999999999.99)); // 少数2桁を維持できない

```
const decimal = fig => {
    // 15桁の壁
    const TOTAL_SAFE_DIGITS = 15;
    const integerDigits = TOTAL_SAFE_DIGITS - fig;
    
    if (integerDigits < 0) throw new Error("fig digits too large");

    // 10進数的な見た目の限界値
    const max = Number("9".repeat(integerDigits) + "." + "9".repeat(fig));
    const min = -max;
    
    return {
        min,
        max,
        valid(v) {
            const absV = Math.abs(v);
            // 1. 15桁の範囲内か
            if (absV > this.max) return false;
            
            // 2. 2進数として完璧に割り切れる（誤差ゼロの）小数か？
            // 52ビットシフトして整数になれば、内部に1ビットの表現誤差も存在しないことが「保証」される
            return Number.isInteger(v * (2 ** 52));
        }
    };
};

// --- 実験 ---
const d = decimal(2);

// 0.25 は 1/4 なので、内部誤差は完全にゼロ。15桁以内。
console.assert(true === d.valid(123.25)); 

// 0.99 は 10進数では綺麗だが、2進数では無限小数（ゴミあり）なので、即座に弾かれる！
console.assert(false === d.valid(99999.99)); 
```





Number型でも条件次第では誤差を完全に排除できるわけか。但しその場合、汎用的な計算はできなさそうだな。2のマイナス乗以外の結果になる算術計算が実行できてしまうのだから。それを制限することはできまい。せっかく以下のような判定ができるのに。

```
// 値を「2のN乗」倍したときに、完全に整数になれば、それは2進数で割り切れる（誤差ゼロの）数。
// 倍精度浮動小数点数の仮数部は52ビットなので、52乗して整数になるかどうかで完璧に判定可能。
const isExactBinaryFraction = v => Number.isInteger(v * (2 ** 52));

class FloatCounter {
    constructor(e) { // e: 1〜52 までの整数値
        // 1. e の範囲チェック (最大52)
        if (!(Number.isSafeInteger(e) && (1 <= e && e <= 52))) {
            throw new RangeError(`eは1〜52までの整数値であるべきです。指定値: ${e}`);
        }

        const scale = 2 ** e;
        // 2. 2進数空間における、絶対に誤差の出ない最大値・最小値を算出
        const max = Number.MAX_SAFE_INTEGER / scale;
        const min = Number.MIN_SAFE_INTEGER / scale;

        this._ = {
            v: 0, 
            resolution: 1 / scale, 
            min, 
            max
        };
    }

    get min() { return this._.min; }
    get max() { return this._.max; }
    get v() { return this._.v; }

    // 値が安全な範囲内（誤差が絶対に出ない領域）に収まっているかを判定
    valid(v) { 
        return this._.min <= v && v <= this._.max; 
    }

    up() {
        const nextV = this._.v + this._.resolution;
        if (!this.valid(nextV)) { 
            throw new RangeError(`値範囲超過 (max: ${this.max})`); 
        }
        this._.v = nextV;
        return this._.v;
    }

    down() {
        const nextV = this._.v - this._.resolution;
        if (!this.valid(nextV)) { 
            throw new RangeError(`値範囲超過 (min: ${this.min})`); 
        }
        this._.v = nextV;
        return this._.v;
    }
}

```javascript
class FloatCounter {
    static Overflow = Object.freeze({
        throw: (self, isDown, nextV)=>{throw new RangeError(`値範囲超過。期待値: ${self.min}〜${self.max} 実際値:${nextV}`);},
        ignore: (self, isDown, nextV)=>self._.v,
        zero: (self, isDown, nextV)=>(self._.v = 0),
        reverse: (self, isDown, nextV)=>(self._.v = (isDown ? self.max : self.min)),
    });
    constructor(e, over=FloatCounter.Overflow.throw) { // e: 1〜52 までの整数値
        // 1. e の範囲チェック (最大52)
        if (!(Number.isSafeInteger(e) && (1 <= e && e <= 52))) {
            throw new RangeError(`eは1〜52までの整数値であるべきです。指定値: ${e}`);
        }

        const scale = 2 ** e;
        // 2. 2進数空間における、絶対に誤差の出ない最大値・最小値を算出
        const max = Number.MAX_SAFE_INTEGER / scale;
        const min = Number.MIN_SAFE_INTEGER / scale;

        if (!Object.values(FloatCounter.Overflow).some(f=>f===over)) {throw new Error(`overはFloatCounter.Overflow.throw,ignore,zero,reverseのいずれかであるべきです。`)}

        this._ = {
            v: 0, 
            resolution: 1 / scale, 
            min, 
            max,
            over,
        };
    }

    get min() { return this._.min; }
    get max() { return this._.max; }
    get resolution() { return this._.resolution; }
    get isIgnore() { return this._.isIgnore; }
    get v() { return this._.v; }

    // 値が安全な範囲内（誤差が絶対に出ない領域）に収まっているかを判定
    valid(v) {return this._.min <= v && v <= this._.max;}
    #set(isDown=false) {
        const nextV = this._.v + (this._.resolution * (isDown ? -1 : 1));
        return this.valid(nextV) ? (this._.v = nextV) : this._.over(this, isDown, nextV);
    }
    up() {return #set()}
    down() {return #set(true)}
}
```

class FloatCounter {
    static Overflow = Object.freeze({
        throw: (self, isDown, nextV) => { 
            throw new RangeError(`値範囲超過。期待値: ${self.min}〜${self.max} 実際値:${nextV}`); 
        },
        ignore: (self, isDown, nextV) => self._.v,
        zero: (self, isDown, nextV) => self._.v = 0,
        // reverse: 下限を下回ったら上限へ、上限を上回ったら下限へワープ
        reverse: (self, isDown, nextV) => self._.v = isDown ? self.max : self.min,
    });

    constructor(e, over = FloatCounter.Overflow.throw) {
        if (!(Number.isSafeInteger(e) && (1 <= e && e <= 52))) {
            throw new RangeError(`eは1〜52までの整数値であるべきです。指定値: ${e}`);
        }

        if (!Object.values(FloatCounter.Overflow).includes(over)) {
            throw new Error(`overはFloatCounter.Overflowのいずれかであるべきです。`);
        }

        const scale = 2 ** e;
        const max = Number.MAX_SAFE_INTEGER / scale;
        const min = Number.MIN_SAFE_INTEGER / scale;

        this._ = {
            v: 0, 
            resolution: 1 / scale, 
            min, 
            max,
            over,
        };
    }

    get min() { return this._.min; }
    get max() { return this._.max; }
    get resolution() { return this._.resolution; }
    get isIgnore() { return this._.over === FloatCounter.Overflow.ignore; }
    get v() { return this._.v; }

    valid(v) { return this._.min <= v && v <= this._.max; }

    // プライベートメソッド
    #set(isDown = false) {
        const nextV = this._.v + (this._.resolution * (isDown ? -1 : 1));
        // validなら値を更新、範囲外なら各overflow関数を実行してその結果を返却
        return this.valid(nextV) ? (this._.v = nextV) : this._.over(this, isDown, nextV);
    }

    up() { return this.#set(); }
    down() { return this.#set(true); }
}
