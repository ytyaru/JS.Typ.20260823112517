構造化を実装する方法を模索する。

```javascrpit
const p = mkFOR(
    'Primitive', 'p', // 短縮名 FORは短縮名が必要 FunctionObjectRoot/Node/Leaf
    v => 'bln int fin big str sym'.split(' ').some(n => p[n](v)),
    {
        bln: mkFOL('Boolean', v => 'boolean' === typeof v),
        int: mkFOL('Integer', v => Number.isSafeInteger(v)),
        fin: mkFOL('Finite', v => Number.isFinite(v) && v <= Number.MAX_SAFE_INTEGER && Number.MIN_SAFE_INTEGER <= v),
        big: mkFOL('BigInt', v => 'bigint' === typeof v),
        str: mkFOL('String', v => 'string' === typeof v),
        sym: mkFOL('Symbol', v => 'symbol' === typeof v),
    }
);
```

```javascrpit
const d = mkFOR(
    'Danger', 'd', // 短縮名 FONは短縮名が必要
    v => 'p o'.split(' ').some(n => d[n](v)),
    {
        p: mkFON('Primitive', 'p',
            v => 'und nul'.split(' ').some(n => d[n](v)) || 'num obj'.split(' ').some(n => d[n].some(v)),
            {
            und: mkFOL('Undefined', v => undefined === v),
            nul: mkFOL('Null', v => null === v),
            num: mkFON('Number', v=>'nan inf over'.split(' ').some(n=>d.p.num[n](v))
                {
                nan: mkFOL('NaN', v => Number.isNaN(v)),
                inf: mkFON('Infinity',v=>[Infinity, -Infinity].some(x => x === v),{
                    p: mkFOL('Positive', v => Infinity === v),
                    n: mkFOL('Negative', v => -Infinity === v),
                }),
                over: mkFON('Over',v=>,{
                    int: mkFOL('Integer', v=>),
                    flt: mkFOL('Float', v=>),
                })
            }),
        }),
        o: mkFON('Object', v=>{
            const N = tof(v);
            return N.startsWith(`BoxedPrimitive<`) || 'NonePrototypeObject PrototypedObject'.split(' ').some(n => n === N);
            }, {
                box: mkFON('BoxedPrimitive', v=>, {
                }),
                none: ,
                proto: 
        }),
    }
);
```
