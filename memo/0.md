# JSにおける簡単な型チェックAPI

```javascript
// Type danger is
Tyd.is.some(v); // 以下どれか
Tyd.is.und(v); // Undefined
Tyd.is.nul(v); // Null
Tyd.is.nan(v); // NaN
Tyd.is.inf(v); // 無限
Tyd.is.pinf(v); // 正無限
Tyd.is.ninf(v); // 負無限
Tyd.is.num(v); // Number: 非数や無限数（NaN,Infinity,-Infinity）を含む。
Tyd.is.fin(v); // Number: 有限数のみ。
Tyd.is.box(v); // BoxedPrimitive（new Boolean | Number | String の3インスタンス）
Tyd.is.nob(v); // Prototypeが無いObject（Object.create(null)で作成された等）
Tyd.is.prototyped(v); // Prototype拡張された非PlainObject（Object.create({}), new Fn()で作成された等）

// Type danger error
Tyd.er.some(v); // 以下どれか
Tyd.er.und(v); // Undefined
Tyd.er.nul(v); // Null
Tyd.er.nan(v); // NaN
Tyd.er.inf(v); // 無限
Tyd.er.pinf(v); // 正無限
Tyd.er.ninf(v); // 負無限
Tyd.er.num(v); // Number: 非数や無限数（NaN,Infinity,-Infinity）を含む。
Tyd.er.fin(v); // Number: 有限数のみ。
Tyd.er.box(v); // BoxedPrimitive（new Boolean | Number | String の3インスタンス）
Tyd.er.nob(v); // Prototypeが無いObject（Object.create(null)で作成された等）
Tyd.er.prototyped(v); // Prototype拡張された非PlainObject（Object.create({}), new Fn()で作成された等）

// Type Primitive is
Typ.is.some(v); // 以下どれか
Typ.is.bln(v); // Boolean: 
Typ.is.int(v); // Number: 安全な整数のみ。Number.isSafeInteger()
Typ.is.big(v); // BigInt: 
Typ.is.str(v); // String: 
Typ.is.sym(v); // Symbol: 
// Type Primitive throw error
Typ.er.some(v); // 以下どれか
Typ.er.bln(v); // Boolean: 
Typ.er.int(v); // Number: 安全な整数のみ。Number.isSafeInteger()
Typ.er.big(v); // BigInt: 
Typ.er.str(v); // String: 
Typ.er.sym(v); // Symbol: 

// Type Object is
Tyo.is.some(v); // どれか
Tyo.is.obj(v); // Pure Object（クラスでもインスタンスでもディスクリプタでも関数でもない）
Tyo.is.ary(v); // Array
Tyo.is.cls(v); // Class Object ES6（ES5は対象外）
Tyo.is.ins(v); // Instance Object
Tyo.is.des(v); // Descriptor Object（getter/setter, value全パターン対象）
Tyo.is.fn(v); // Function Object（`function`,Arrow,bind,組込も含む全callable）
Tyo.is.md(v); // Method Object（Object, Class, Instance）

// Type Object error
Tyo.er.some(v); // どれか
Tyo.er.obj(v); // Pure Object
Tyo.er.ary(v); // Array
Tyo.er.cls(v); // Class Object ES6（ES5は対象外）
Tyo.er.ins(v); // Instance Object
Tyo.er.des(v); // Descriptor Object
Tyo.er.fn(v); // Function Object（`function`,Arrow,bind,組込も含む全callable）
Tyo.er.md(v); // Method Object（Object, Class, Instance）

// Type Arrayable Primitive is    fn: (v,i)=>0<v のように任意関数で各値のバリデーションを書ける

Typa.er.big(v, fn); // BigInt:
```
